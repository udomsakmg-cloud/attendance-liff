async function submitCheckIn() { const btn = document.getElementById('btnSubmit'); const originalBtnText = btn.innerText; // ------------------------------------------------------------- // Step 1: [Optimistic UI] เปลี่ยนสถานะหน้าจอทันที ไม่ต้องรอเซิร์ฟเวอร์ // ------------------------------------------------------------- btn.disabled = true; btn.innerText = 'กำลังบันทึก...'; // แสดง Toast / Pop-up สำเร็จล่วงหน้าทันที (ตัวอย่างใช้ SweetAlert2) Swal.fire({ icon: 'success', title: 'ลงเวลาสำเร็จ!', text: 'ระบบกำลังบันทึกข้อมูลของคุณ...', timer: 1500, showConfirmButton: false }); // รวบรวมข้อมูลสำหรับส่ง const payload = { type: document.getElementById('checkType').value, name: currentUserDisplayName, // ชื่อจาก LIFF lat: currentLat, lng: currentLng, image: compressedBase64Image // รูปภาพที่บีบอัดแล้ว }; // ------------------------------------------------------------- // Step 2: ส่งข้อมูลไป Google Apps Script เบื้องหลัง (Background Fetch) // ------------------------------------------------------------- try { const response = await fetch('YOUR_GAS_WEB_APP_URL', { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify(payload) }); const result = await response.json(); // ------------------------------------------------------------- // Step 3: ตรวจสอบผลลัพธ์จริงจาก Apps Script // ------------------------------------------------------------- if (result.success) { // หากสำเร็จ ปิดหน้าต่าง LIFF ทันที setTimeout(() => { if (typeof liff !== 'undefined' && liff.isInClient()) { liff.closeWindow(); } }, 800); } else { // ถ้าหลังบ้านตอบกลับมาว่าไม่สำเร็จ (เช่น ติด Cooldown 30 นาที หรือ ระยะทางเกิน) throw new Error(result.message); } } catch (error) { // ------------------------------------------------------------- // Step 4: [Rollback] หากเกิดข้อผิดพลาด ให้คืนค่าหน้าจอกลับมา // ------------------------------------------------------------- Swal.fire({ icon: 'warning', title: 'ไม่สามารถบันทึกได้', text: error.message || 'เกิดข้อผิดพลาดในการเชื่อมต่อ', confirmButtonText: 'ตกลง' }); // คืนค่าปุ่มให้กดใหม่ได้ btn.disabled = false; btn.innerText = originalBtnText; } }
